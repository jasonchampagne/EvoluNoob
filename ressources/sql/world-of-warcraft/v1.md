# World of Warcraft - données (v1)

```sql
-- ----------------------------------------------------------------------------------------------------------------------------
-- World of Warcraft (v1)
-- ----------------------------------------------------------------------------------------------------------------------------

-- Quelques domaines (types métier)
CREATE DOMAIN level_t AS SMALLINT CHECK (VALUE BETWEEN 1 AND 90);
CREATE DOMAIN gold_t AS INTEGER CHECK (VALUE >= 0);
CREATE DOMAIN rarity_t AS VARCHAR(20) 
    CHECK (VALUE IN ('Mauvais','Commun','Inhabituel','Rare','Épique','Légendaire','Prodigieuse','Héritage'));
CREATE DOMAIN gender_t AS CHAR(1) CHECK (VALUE IN ('H','F'));

-- Factions
CREATE TABLE FACTIONS
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(20) NOT NULL UNIQUE CHECK (name IN ('Alliance','Horde','Neutre')),
    type VARCHAR(20) NOT NULL
);

-- Races
CREATE TABLE RACES
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(30) NOT NULL UNIQUE,
    faction_affinity_fkid INTEGER REFERENCES FACTIONS(id)
);

-- Classes
CREATE TABLE CLASSES
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(30) NOT NULL UNIQUE
);

-- Spécialisations
CREATE TABLE SPECIALIZATIONS
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    class_fkid INTEGER NOT NULL REFERENCES CLASSES(id),
    name VARCHAR(40) NOT NULL,
    role VARCHAR(10) NOT NULL CHECK (role IN ('Tank','Soigneur','DPS')),
    UNIQUE (class_fkid, name)
);

-- Guildes
CREATE TABLE GUILDS
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(40) NOT NULL UNIQUE,
    faction_affiliation_fkid INTEGER NOT NULL REFERENCES FACTIONS(id),
    founded_at DATE NOT NULL DEFAULT CURRENT_DATE
);

-- Personnages
CREATE TABLE CHARACTERS
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    username VARCHAR(12) NOT NULL UNIQUE,
    race_fkid INTEGER NOT NULL REFERENCES RACES(id),
    specialization_fkid INTEGER NOT NULL REFERENCES SPECIALIZATIONS(id),
    gender gender_t NOT NULL,
    level level_t NOT NULL,
    guild_fkid INTEGER REFERENCES GUILDS(id) ON DELETE SET NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CHECK (level BETWEEN 1 AND 90)
);

-- Groupes
CREATE TABLE GROUPS
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(40),
    leader_character_fkid INTEGER REFERENCES CHARACTERS(id)
);

-- Quêtes
CREATE TABLE QUESTS
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    title VARCHAR(80) NOT NULL,
    required_level level_t NOT NULL,
    zone VARCHAR(40) NOT NULL,
    gold_reward gold_t NOT NULL DEFAULT 0,
    xp_reward INTEGER NOT NULL DEFAULT 0,
    prerequisite_quest_fkid INTEGER REFERENCES QUESTS(id)
);

-- Donneurs de quêtes
CREATE TABLE CHARACTER_QUESTS
(
    character_fkid INTEGER NOT NULL REFERENCES CHARACTERS(id) ON DELETE CASCADE,
    quest_fkid INTEGER NOT NULL REFERENCES QUESTS(id) ON DELETE CASCADE,
    status VARCHAR(10) NOT NULL CHECK (status IN ('En cours','Réussie','Échouée')),
    completed_at DATE,
    PRIMARY KEY (character_fkid, quest_fkid)
);

-- Objets
CREATE TABLE ITEMS
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(60) NOT NULL,
    rarity rarity_t NOT NULL,
    item_level INTEGER NOT NULL CHECK (item_level BETWEEN 1 AND 999),
    sell_price gold_t NOT NULL
);

-- Inventaires
CREATE TABLE INVENTORIES
(
    character_fkid INTEGER NOT NULL REFERENCES CHARACTERS(id) ON DELETE CASCADE,
    item_fkid INTEGER NOT NULL REFERENCES ITEMS(id),
    quantity INTEGER NOT NULL CHECK (quantity > 0),
    PRIMARY KEY (character_fkid, item_fkid)
);

-- Statistiques d'objets
CREATE TABLE ITEM_STATS
(
    item_fkid INTEGER PRIMARY KEY REFERENCES ITEMS(id) ON DELETE CASCADE,
    details JSON NOT NULL
);

-- Instances (donjons ou raids)
CREATE TABLE INSTANCES
(
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(60) NOT NULL UNIQUE,
    min_level level_t NOT NULL,
    max_level level_t NOT NULL,
    type VARCHAR(10) NOT NULL CHECK (type IN ('Donjon','Raid')),
    CHECK (min_level <= max_level)
);

-- Historique des instances
CREATE TABLE INSTANCE_RUNS
(
    instance_fkid INTEGER NOT NULL REFERENCES INSTANCES(id),
    group_fkid INTEGER NOT NULL REFERENCES GROUPS(id),
    run_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    succeeded BOOLEAN NOT NULL,
    PRIMARY KEY (instance_fkid, group_fkid, run_at)
);

-- Butins d'instances
CREATE TABLE INSTANCE_LOOTS
(
    instance_fkid INTEGER NOT NULL REFERENCES INSTANCES(id) ON DELETE CASCADE,
    item_fkid INTEGER NOT NULL REFERENCES ITEMS(id) ON DELETE CASCADE,
    drop_rate NUMERIC(5,2) NOT NULL CHECK (drop_rate BETWEEN 0 AND 100),
    PRIMARY KEY (instance_fkid, item_fkid)
);

-- Relations entre joueurs
CREATE TABLE FRIENDSHIPS
(
    source_fkid INTEGER NOT NULL REFERENCES CHARACTERS(id) ON DELETE CASCADE,
    target_fkid INTEGER NOT NULL REFERENCES CHARACTERS(id) ON DELETE CASCADE,
    PRIMARY KEY (source_fkid, target_fkid),
    CHECK (source_fkid <> target_fkid)
);
```
